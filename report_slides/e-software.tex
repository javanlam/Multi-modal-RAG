\section{Required Hardware and Software}

\subsection{Hardware}
\textbf{Development PC}: a Unix-based system.

\subsection{Software}

The following Python packages are required. 

\subsubsection*{Python Dependencies}
\noindent
\begin{itemize}
    \item \texttt{pandas}, \texttt{numpy}: data handling and numerical operations
    \item \texttt{PyMuPDF}, \texttt{python-docx}: PDF and DOCX file processing
    \item \texttt{torch}, \texttt{torchvision}, \texttt{Pillow}: deep learning and image processing
    \item \texttt{sentence-transformers}, \texttt{transformers}: text embeddings and transformer models
    \item \texttt{langchain-text-splitters}: text chunking for retrieval
    \item \texttt{chromadb}: vector database
    \item \texttt{networkx}, \texttt{python-louvain}: knowledge graphs and community detection
    \item \texttt{openai}, \texttt{google-generativeai}, \texttt{dashscope}: access to GenAI models
    \item \texttt{DINOv2} dependencies: advanced image feature extraction
            \begin{itemize}
                \item \texttt{torch}, \texttt{torchvision}, \texttt{torchmetrics}
                \item \texttt{omegaconf}
                \item \texttt{fvcore}
                \item \texttt{iopath}
                \item \texttt{xformers}
                \item \texttt{cuml-cu11}
                \item \texttt{submitit}
            \end{itemize}
    \item \texttt{Talk2DINO} dependencies: vision-language cross-modal alignment and interaction
            \begin{itemize}
                \item \texttt{matplotlib}
                \item \texttt{opencv-python}
                \item \texttt{scikit-image}
                \item \texttt{timm}
                \item \texttt{pyyaml}
                \item \texttt{requests}
                \item \texttt{tqdm}
                \item \texttt{einops}
                \item \texttt{webdataset}
                \item \texttt{numpy}
            \end{itemize}
    \item \texttt{CLIP}: multimodal embeddings
\end{itemize}
